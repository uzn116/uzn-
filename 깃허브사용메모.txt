vscode 터미널에  깃허브, 깃허브페이지 연결하기, 

 git --version  
# 다운로드: Git 공식 홈페이지(git-scm.com)에 접속
git version 2.xx.x와 같이 숫자가 잘 나오는지 확인

git init
# 이라는 명령어를 사용하면 이 폴더 안의 모든 파일들을 Git이 추적하기 시작합니다. 터미널에 다음을 입력해 보세요.

git add .

git commit -m "첫 번째 코드 저장"                              이름은 바꿔도 됨 엔터 

git config --global user.email "내_이메일@example.com"                  내 이메일로 엔터 

git config --global user.name "내_이름"                                  내 이름으로 엔터 


git commit -m "첫 번째 코드 저장"   
#커밋이 성공적으로 완료되었다고 터미널에 메시지가 뜨는지 확인


깃허브가입 연결 / 웹주소 복사 

git remote add origin https://github.com/uzn116/uzn-.git                            엔터 

git push -u origin master                                                                  엔터 
# 로그인 또는 패스워드
